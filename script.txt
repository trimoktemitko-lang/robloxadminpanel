local function JLKY_fake_script() -- autokillertp.LocalScript 
	local script = Instance.new('LocalScript', autokillertp)

	local Players = game:GetService("Players")
	local LocalPlayer = Players.LocalPlayer
	local button = script.Parent
	
	local running = false
	local killerLoop = nil
	
	-- Notification helper
	local function showNotification(msg)
	    local notif = Instance.new("TextLabel")
	    notif.Size = UDim2.new(1, 0, 0, 24)
	    notif.Position = UDim2.new(0, 0, 0, -24)
	    notif.BackgroundTransparency = 0.5
	    notif.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
	    notif.TextColor3 = Color3.fromRGB(255, 0, 0)
	    notif.Font = Enum.Font.SourceSansBold
	    notif.Text = msg
	    notif.TextScaled = true
	    notif.Parent = button
	    task.wait(2)
	    notif:Destroy()
	end
	
	local function teleportAndKill()
	    while running do
	        local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
	        local humanoid = character:FindFirstChild("Humanoid")
	        if not humanoid or humanoid.Health <= 0 then
	            task.wait(1)
	            continue
	        end
	
	        -- Find a target player to kill
	        local targetPlayer = nil
	        for _, player in Players:GetPlayers() do
	            if player ~= LocalPlayer then
	                local targetChar = player.Character
	                local targetHumanoid = targetChar and targetChar:FindFirstChild("Humanoid")
	                if targetChar and targetHumanoid and targetHumanoid.Health > 0 then
	                    targetPlayer = player
	                    break
	                end
	            end
	        end
	
	        if targetPlayer and targetPlayer.Character then
	            local targetChar = targetPlayer.Character
	            local targetHumanoid = targetChar:FindFirstChild("Humanoid")
	            if targetHumanoid and targetHumanoid.Health > 0 then
	                -- Teleport local player behind the target
	                local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
	                local localRoot = character:FindFirstChild("HumanoidRootPart")
	                if targetRoot and localRoot then
	                    -- Calculate position 2 studs behind and 2 studs above the target
	                    local behindOffset = targetRoot.CFrame.LookVector * -2 -- 2 studs behind
	                    local aboveOffset = Vector3.new(0, 2, 0) -- 2 studs above
	                    local newCFrame = targetRoot.CFrame + behindOffset + aboveOffset
	                    localRoot.CFrame = newCFrame
	                end
	
	                -- Simulate autoclick: try to damage the target humanoid
	                for _, part in targetChar:GetChildren() do
	                    if part:IsA("BasePart") then
	                        part:BreakJoints() -- This will "kill" the target on client, but may not replicate to server
	                        break
	                    end
	                end
	            end
	        else
	        end
	
	        task.wait(0) -- Wait before next kill attempt
	    end
	end
	
	-- Teleport to a random alive player when local player dies
	local function onCharacterAdded(character)
	    local humanoid = character:FindFirstChild("Humanoid")
	    if humanoid then
	        humanoid.Died:Connect(function()
	            -- Find all alive players except LocalPlayer
	            local alivePlayers = {}
	            for _, player in Players:GetPlayers() do
	                if player ~= LocalPlayer then
	                    local targetChar = player.Character
	                    local targetHumanoid = targetChar and targetChar:FindFirstChild("Humanoid")
	                    if targetChar and targetHumanoid and targetHumanoid.Health > 0 then
	                        table.insert(alivePlayers, player)
	                    end
	                end
	            end
	
	            if #alivePlayers > 0 then
	                -- Pick a random player
	                local idx = math.random(1, #alivePlayers)
	                local chosenPlayer = alivePlayers[idx]
	                local chosenChar = chosenPlayer.Character
	                local chosenRoot = chosenChar and chosenChar:FindFirstChild("HumanoidRootPart")
	                local localRoot = character:FindFirstChild("HumanoidRootPart")
	                if chosenRoot and localRoot then
	                    -- Teleport local player to the chosen player's position (2 studs above)
	                    local newCFrame = chosenRoot.CFrame + Vector3.new(0, 2, 0)
	                    localRoot.CFrame = newCFrame
	                    showNotification("Teleported to " .. chosenPlayer.Name)
	                end
	            else
	                showNotification("No alive players to teleport to!")
	            end
	        end)
	    end
	end
	
	-- Listen for character added to bind Died event
	LocalPlayer.CharacterAdded:Connect(onCharacterAdded)
	-- Also bind if character already exists
	if LocalPlayer.Character then
	    onCharacterAdded(LocalPlayer.Character)
	end
	
	button.MouseButton1Click:Connect(function()
	    running = not running
	    if running then
	        button.BackgroundColor3 = Color3.fromRGB(0, 0, 255) -- Blue when running
	        killerLoop = coroutine.create(teleportAndKill)
	        coroutine.resume(killerLoop)
	    else
	        button.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
	    end
	end)
	
	
end

local function HWAYP_fake_script() -- Fly.Fly_704f687 
	local script = Instance.new('LocalScript', Fly)

	local Players = game:GetService("Players")
	local LocalPlayer = Players.LocalPlayer
	local button = script.Parent
	
	local flying = false
	local speed = 50
	
	local bodyVelocity = nil
	local bodyGyro = nil
	
	local function enableFly()
		local character = LocalPlayer.Character
		if not character then
			character = LocalPlayer.CharacterAdded:Wait()
		end
		local root = character:FindFirstChild("HumanoidRootPart")
		local humanoid = character:FindFirstChildOfClass("Humanoid")
		if not root or not humanoid then return end
	
		bodyVelocity = Instance.new("BodyVelocity")
		bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		bodyVelocity.Velocity = Vector3.new(0, 0, 0)
		bodyVelocity.Parent = root
	
		bodyGyro = Instance.new("BodyGyro")
		bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
		bodyGyro.P = 9e4
		bodyGyro.CFrame = root.CFrame
		bodyGyro.Parent = root
	
		-- Flight loop — uses MoveDirection only (works with keyboard WASD and mobile joystick)
		coroutine.wrap(function()
			while flying do
				local character = LocalPlayer.Character
				if not character then break end
				local root = character:FindFirstChild("HumanoidRootPart")
				local humanoid = character:FindFirstChildOfClass("Humanoid")
				if not root or not humanoid then break end
				if not bodyVelocity or not bodyVelocity.Parent then break end
				if not bodyGyro or not bodyGyro.Parent then break end
				while flying do
					humanoid.PlatformStand = true
					task.wait()
				end
				local cam = workspace.CurrentCamera
				local camCF = cam.CFrame
				local camLookH = (camCF.LookVector * Vector3.new(1, 0, 1)).Unit
				local camRightH = (camCF.RightVector * Vector3.new(1, 0, 1)).Unit
	
				local moveDir = humanoid.MoveDirection
	
				if moveDir.Magnitude > 0 then
					-- Decompose MoveDirection into forward/back and left/right relative to camera
					local forwardBack = moveDir:Dot(camLookH)
					local leftRight = moveDir:Dot(camRightH)
	
					-- Forward/back uses full 3D camera LookVector (look up + forward = fly up)
					-- Left/right uses horizontal camera RightVector (strafe)
					local moveVec = camCF.LookVector * forwardBack + camCF.RightVector * leftRight
	
					bodyVelocity.Velocity = moveVec * speed
				else
					bodyVelocity.Velocity = Vector3.new(0, 0, 0)
				end
	
				bodyGyro.CFrame = camCF
				task.wait()
			end
	
			if flying then
				disableFly()
				button.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
			end
		end)()
	end
	
	local function disableFly()
		flying = false
	
		local character = LocalPlayer.Character
		if character then
			local root = character:FindFirstChild("HumanoidRootPart")
			if root then
				if bodyVelocity then bodyVelocity:Destroy() bodyVelocity = nil end
				if bodyGyro then bodyGyro:Destroy() bodyGyro = nil end
			end
		end
	end
	
	button.MouseButton1Click:Connect(function()
		flying = not flying
		if flying then
			enableFly()
			button.BackgroundColor3 = Color3.fromRGB(0, 0, 255)
		else
			disableFly()
			button.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
		end
	end)
	
	LocalPlayer.CharacterAdded:Connect(function()
		if flying then
			enableFly()
		end
	end)
	
	
end

	                    killed = true
	                    break
	                end
	            end
	            if killed then
	            else
	            end
	        else
	        end
	
	        task.wait() -- Wait before next kill attempt
	    end

	end
